## Save your first preferences

```bash
pnpm add @ez-kit/va-store valtio
```

```tsx
'use client'

import { createContextStore, pipe, StoreProvider } from '@ez-kit/va-store'
import { withPersist } from '@ez-kit/va-store/persist'
import { localStorageAdapter } from '@ez-kit/va-store/persist/storage'
import { proxy } from 'valtio'

type Preferences = { theme: 'light' | 'dark'; fontScale: number }

const sources = { browser: localStorageAdapter({ storageKey: 'my-app-preferences' }) }

const prefs = createContextStore(
	() =>
		pipe(
			proxy<Preferences>({ theme: 'light', fontScale: 1 }),
			withPersist({
				fields: (field) => [field((state) => state.theme, {}), field((state) => state.fontScale, {})],
			}),
		),
	{ name: 'preferences' },
)

function ThemeButton() {
	const store = prefs.useStore()
	const { theme } = prefs.useSnapshot()
	return (
		<button
			type='button'
			onClick={() => {
				store.theme = theme === 'light' ? 'dark' : 'light'
			}}
		>
			Theme: {theme}
		</button>
	)
}

export default function PreferencesPage() {
	return (
		<StoreProvider
			persist={sources}
			defaultSource='browser'
		>
			<prefs.Provider>
				<ThemeButton />
			</prefs.Provider>
		</StoreProvider>
	)
}
```

Change **Theme** and **Font scale**, then reload this page. Both values restore automatically
from browser storage. Use the example’s file tabs to see the provider, store and controls in separate files.

The provider registers a source named `browser`. The store selects the fields to save.
An empty field configuration uses `defaultSource`. A stable store name and storage key let
a new instance find the same saved values after reload.
